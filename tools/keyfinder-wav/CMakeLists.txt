cmake_minimum_required(VERSION 3.16)
project(keyfinder-cli LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)

add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../libkeyfinder" "${CMAKE_BINARY_DIR}/libkeyfinder")
add_executable(keyfinder-cli main.cpp)
target_include_directories(keyfinder-cli PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../libkeyfinder/src")
target_link_libraries(keyfinder-cli PRIVATE keyfinder)
